# 加密解密

### MD5加密 — MD5Encode(text)

**线上实际示例 — 生成接口签名：**

```javascript
var AbstractTrigger = require('AbstractTrigger');
class MyTrigger extends AbstractTrigger {
  execute(context, param) {
    var timestamp = new Date().getTime();
    var access_token = MD5Encode(timestamp + '1ac7208b21c34eecac230cf7cba64306');
    var url = 'http://zt.oa.amywechat.com/api/v1/receiveApi/modifyOrder?timestamp=' + timestamp + '&access_token=' + access_token;
    var saveHeader = { 'Content-Type': 'application/json; charset=utf-8' };
    var saveBody = { orderSno: param.return.code };
    var sendRes = JSON.parse(postman('post', url, JSON.stringify(saveHeader), JSON.stringify(saveBody)));
    return {};
  }
}
exports({ entryPoint: MyTrigger });
```

### Base64编码/解码 — Base64Encode(text) / Base64Decode(text)

**线上实际示例 — Basic认证编码：**

```javascript
var AbstractTrigger = require('AbstractTrigger');
class MyTrigger extends AbstractTrigger {
  execute(context, param) {
    var resAccessToken = Base64Encode('clientId:clientSecret');
    var headerToken = { Authorization: 'Basic ' + resAccessToken };
    var strResponseToken = postman('POST', 'https://api.huilianyi.com/gateway/oauth/token?grant_type=client_credentials&scope=write', JSON.stringify(headerToken), JSON.stringify({}));
    return { strResponseToken: JSON.parse(strResponseToken) };
  }
}
exports({ entryPoint: MyTrigger });
```

**线上实际示例 — Base64解码解析许可证：**

```javascript
var content = result[0].content;
var datas = JSON.parse(split(content, ';', 4));
var param1 = Base64Decode(datas[0]);
var param2 = Base64Decode(datas[1]);
var param3 = MD5Encode(expireTime);
var param4 = MD5Encode(tenantId);
```

### AES加密/解密 — AESEncrypt(text, key, iv) / AESDecrypt(text, key, iv)

**线上实际示例 — 加密接口返回数据：**

```javascript
var AbstractAPIHandler = require('AbstractAPIHandler');
class MyAPIHandler extends AbstractAPIHandler {
  execute(request) {
    var key = 'pyA819tUGF0glrrS';
    var iv = 'pyA819tUGF0glrrS';
    var env = ObjectStore.env();
    var url = env.url + '/iuap-api-gateway/yonbip/hrcloud/payfile/queryWaPayfileDetailByCond?schemeCode=' + request.code + '&payPeriod=' + request.payPeriod;
    var infoString = ObjectStore.openLinker('GET', url, 'AT1DB80D3452600008', {});
    return { plan: AESEncrypt(infoString, key, iv) };
  }
}
exports({ entryPoint: MyAPIHandler });
```

**线上实际示例 — AES解密：**

```javascript
var encrypted = AESEncrypt(10325, '0123456789abcdef0123456789abcdef', '1234567890123456');
var decrypted = AESDecrypt('t8r7srV4jLsDU/RzkdhTnQ==', '0123456789abcdef0123456789abcdef', '1234567890123456');
```

### DES加密/解密 — DESEncrypt(text, key) / DESDecrypt(text, key)

**线上实际示例 — 加密时间戳：**

```javascript
var AbstractAPIHandler = require('AbstractAPIHandler');
class MyAPIHandler extends AbstractAPIHandler {
  execute(request) {
    var date = JSON.stringify(new Date().getTime());
    var key = 'DesEncrypt';
    var encrypted = DESEncrypt(date, key);
    return { res: encrypted };
  }
}
exports({ entryPoint: MyAPIHandler });
```

**线上实际示例 — DES解密：**

```javascript
var AbstractAPIHandler = require('AbstractAPIHandler');
class MyAPIHandler extends AbstractAPIHandler {
  execute(request) {
    var date = request.date;
    var key = 'DesEncrypt';
    var res = DESDecrypt(date, key);
    return { res };
  }
}
exports({ entryPoint: MyAPIHandler });
```

### HmacSHA256 — HmacSHA256(text, key)

**线上实际示例：**

```javascript
var AbstractTrigger = require('AbstractTrigger');
class MyTrigger extends AbstractTrigger {
  execute(context, param) {
    var date = 'password';

exports({ entryPoint: MyTrigger });
```

### 其他加密函数

| 函数 | 说明 |
|------|------|
| `Base62Encode(text)` / `Base62Decode(text)` | Base62 编码/解码 |
| `Base64MD5Encode(text)` | MD5编码后再Base64编码 |
| `SHA1(text)` | SHA1加密 |
| `SHA256Encode(text)` | SHA256加密 |
| `HmacSHA(text, key)` | Hmac加密 |
| `HmacSHA256Base64(text, key)` | HmacSHA256加密后Base64编码 |
| `HmacSHA1Base64(text, key)` | HmacSHA1加密后Base64编码 |
| `HEXEncode(text)` / `HEXDecode(text)` | HEX编码/解码 |
| `MurmurHash(text)` | 32位MurmurHash编码 |
| `MurmurHash64A(text)` | 64位MurmurHash编码 |
