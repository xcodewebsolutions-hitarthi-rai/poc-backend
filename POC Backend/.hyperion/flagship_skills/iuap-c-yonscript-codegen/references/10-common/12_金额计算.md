# 金额计算

| 函数 | 说明 |
|------|------|
| `GetBigDecimal(value)` | 转换成BigDecimal数据类型 |
| `AmountPointNum(value)` | 获取小数点位数 |
| `moneyFormat(value, scale?)` | 四舍五入格式化金额，返回BigDecimal类型 |
| `MoneyFormatReturnBd(value, scale)` | 四舍五入格式化金额，返回BigDecimal（常配合 `.set()` 使用） |

### 线上实际示例 — moneyFormat 格式化金额：

```javascript
var AbstractTrigger = require('AbstractTrigger');
class MyTrigger extends AbstractTrigger {
  execute(context, param) {
    var data = param.data[0];
    var sumMoney = 0;
    data.deliveryDetails.map(function (item) {
      var money = Number(ObjectStore.moneyFormat(Number(item.deliveryDetailDefineCharacter.amount)));
      sumMoney = sumMoney + money;
    });
    return {};
  }
}
exports({ entryPoint: MyTrigger });
```

### 线上实际示例 — MoneyFormatReturnBd 在规则链中格式化子表金额：

```javascript
var AbstractTrigger = require('AbstractTrigger');
class MyTrigger extends AbstractTrigger {
  execute(context, param) {
    if (param.requestAction == 'import') {
      var detail = param.data[0].insurancePayRepayTypeDetails;
      if (detail) {
        detail.forEach(function (item) {
          item.set('payCompanyRepayAmount', MoneyFormatReturnBd(item.payCompanyRepayAmount, 2));
          item.set('payPersonalRepayAmount', MoneyFormatReturnBd(item.payPersonalRepayAmount, 2));
        });
      }
    }
    return {};
  }
}
exports({ entryPoint: MyTrigger });
```
