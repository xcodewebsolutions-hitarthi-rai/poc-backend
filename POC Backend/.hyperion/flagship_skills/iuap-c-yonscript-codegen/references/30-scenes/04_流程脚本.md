# 流程脚本

| 功能 | 说明 |
|------|------|
| `workflow-deleteProcess` | 撤回工作流 |
| `eeacSendOrReverse` | 业务事项发送、业务事项逆向回调 |
| `queryCreditBalance` | 信用余额查询 |
| `eeacsend` | 执行成功后自动过账，失败或未执行则不过账 |
| `eeacreverse` | 执行成功后自动取消过账，失败或未执行则不取消 |
| `busflow-push` | 下推业务流 |
| `ObjectStore.execute submit` | 发起工作流 |

### 撤回工作流（deleteWorkFlowByBusinessKey）

通过 businessKey 撤回工作流。

```javascript
var billnum = 'yourbillnum';
var billid = 'yourid';
var businessKey = billnum + '_' + billid;
var result = deleteWorkFlowByBusinessKey(businessKey);
```

### 下推业务流（busWorkFlowPush）

```javascript
var billnum = 'buillnum';
var runleid = 'runid';
var ids = ['id1', 'id2'];
var result = busWorkFlowPush(billnum, runleid, ids);
```

### 发起工作流（ObjectStore.execute submit）

```javascript
var AbstractAPIHandler = require('AbstractAPIHandler');
class MyAPIHandler extends AbstractAPIHandler {
  execute(request) {
    var object = { id: request.id };
    var detail = ObjectStore.selectById('AT15D1B41A05D00004.AT15D1B41A05D00004.flow1020', object);
    var data = { billnum: 'flow1020', data: JSON.stringify(detail) };
    var res = ObjectStore.execute('submit', data);
    return { res };
  }
}
exports({ entryPoint: MyAPIHandler });
```

### 业务事项发送/逆向回调（ObjectStore.eeacSendOrReverse）

`type` 传 `'send'` 发送事项，传 `'reverse'` 逆向回调。

```javascript
var object = { id: '1946459928583471104' };
var selectByIdRes = ObjectStore.selectById('AT19CBD6F405F80005.AT19CBD6F405F80005.test_busifuntion02', object);
var data = {
  billnum: 'test_busifuntion0202List',
  busiObj: 'test_busifuntion02',
  eeacData: selectByIdRes
};
var type = 'send'; // send: 发送事项, reverse: 逆向回调
var res = ObjectStore.eeacSendOrReverse(data, type);
```

### 信用余额查询（queryCreditBalance）

通过 POST 接口查询客户/业务员/组织的信用余额。

```javascript
var agentId = param.variablesMap.agentId; // 客户id
var salesOrgId = param.variablesMap.salesOrgId; // 组织id
var currency = param.variablesMap['orderPrices.currency']; // 币种id
var header = { 'Content-Type': 'application/json;charset=UTF-8' };
var body = {
  billnum: 'credit_recordlist',
  data: [{
    targetObjId: agentId,     // 目标对象ID
    currencyId: currency,     // 币种
    controlOrgId: salesOrgId, // 组织
    targetType: 0,            // 0:客户, 1:业务员, 2:销售组织, 3:部门, 4:对象域
  }],
};
var url = ObjectStore.env().url + '/yonbip-mkt-mkc2b/api/credit/query';
var apiResponse = ObjectStore.postman('post', url, header, body);
if (!apiResponse) {
  throw new Error({ type: 'error', title: '信用余额查询可能超时' });
}
var response = JSON.parse(apiResponse);
if (response.code != 200) {
  throw new Error(response.code + response.message);
}
var creditBalance = response.data[0].creditBalance;
```

### 线上实际示例 — 自动过账（eeacsend）：

执行成功后自动过账，失败或未执行则不过账。

```javascript
var AbstractTrigger = require('AbstractTrigger');
class MyTrigger extends AbstractTrigger {
  execute(context, param) {
    // 重置resubmitCheckKey以免报主键冲突
    var resubmitCheckKey = param.billDOs[0].resubmitCheckKey;
    if (resubmitCheckKey != undefined && resubmitCheckKey.length > 35) {
      resubmitCheckKey = substring(resubmitCheckKey, 0, resubmitCheckKey.length - 35) + uuid();
      param.billDOs[0].set('resubmitCheckKey', resubmitCheckKey);
    }

    var eeacStatus = param.billDOs[0].eeacStatus;
    var content = {
      billnum: param.billDOs[0].cBillNo,
      busiObj: param.billDOs[0].busiObj,
      data: JSON.stringify(param.billDOs[0]),
    };

    if (eeacStatus != undefined && eeacStatus == '0') {
      ObjectStore.execute('eeacsend', content);
      var eeacVersion = Number(param.billDOs[0].eeacVersion) + 1;
      param.billDOs[0].set('eeacStatus', '1');
      param.billDOs[0].set('eeacVersion', eeacVersion + '');
    } else if (eeacStatus != undefined && (eeacStatus == '2' || eeacStatus == '3' || eeacStatus == '4')) {
      ObjectStore.execute('eeacreverse', content);
      ObjectStore.execute('eeacsend', content);
      var eeacVersion = Number(param.billDOs[0].eeacVersion) + 1;
      param.billDOs[0].set('eeacStatus', '1');
      param.billDOs[0].set('eeacVersion', eeacVersion + '');
    }

    return {};
  }
}
exports({ entryPoint: MyTrigger });
```

### 线上实际示例 — 自动取消过账（eeacreverse）：

执行成功后自动取消过账，失败或未执行则不取消。

```javascript
var AbstractTrigger = require('AbstractTrigger');
class MyTrigger extends AbstractTrigger {
  execute(context, param) {
    // 兼容物理删除时框架没有修改_status的场景
    var actionCode = param.baseBillContext.actionInfo.code;
    if (includes(actionCode, 'delete') && param.billDOs[0]._status != 3) {
      param.billDOs[0].set('_status', 3);
    }

    var eeacStatus = param.billDOs[0].eeacStatus;
    var content = {
      billnum: param.billDOs[0].cBillNo,
      busiObj: param.billDOs[0].busiObj,
      data: JSON.stringify(param.billDOs[0]),
    };

    if (eeacStatus != undefined && eeacStatus == '1') {
      throw new Error('过账中的单据不允许删除！');
    } else if (eeacStatus != undefined && (eeacStatus == '2' || eeacStatus == '3' || eeacStatus == '4')) {
      ObjectStore.execute('eeacreverse', content);
    }

    return {};
  }
}
exports({ entryPoint: MyTrigger });
```
