# 三方集成封装规范（YonScript/V8）

> **加载时机**：NCC、SAP、浪潮、金蝶、第三方系统、跨租户接口、统一连接器场景。
> **职责**：定义 V8 环境下三方调用的分层方式、认证配置、错误处理和返回解析。
> **唯一来源**：三方调用封装规则只维护在本文档，远程调用基础 API 见 [../10-common/05_远程调用.md](../10-common/05_远程调用.md)。

---

## 核心边界

| 场景 | API | 说明 |
|---|---|---|
| 本租户 BIP 标准 API | `ObjectStore.openLinker` | 自动认证；URL/入参未知时先查 `iuap-c-openapi-integration` |
| 跨租户 BIP / SAP / 浪潮 / 金蝶 | `ObjectStore.postman` | 手动认证；返回字符串默认 `safeJsonParse` |
| NCC/高级版 | `ObjectStore.nccLinker` | 使用 `nccEnv` 对象 |
| 调用公共脚本 | `extrequire('GTxxx.Module.func')` | 复用 token、连接器、工具函数 |

V8 禁止项：禁止 Java SDK、Python、Node.js 模块、原生 HTTP 客户端、箭头函数；变量声明和 `this.` 调用以 [../00-governance/02_V8沙箱安全模式.md](../00-governance/02_V8沙箱安全模式.md) 为准。

NCC/高级版场景如果现场已提供统一调用脚本（如 `RBSM.APIhs.InvokeNCC`），业务 Rule/API 脚本必须优先复用该封装，只传业务报文和接口 URI；禁止绕过现场封装重新拼 `nccEnv`、`baseUrl` 或直接调用 `ObjectStore.nccLinker`。只有明确不存在现场封装时，才按本文 NCC 连接器差异示例生成基础调用。

---

## 分层模式

```
业务 Rule/API 脚本
  └─ extrequire 调用连接器脚本（可选）
      └─ ObjectStore.postman / ObjectStore.nccLinker
          └─ 三方系统
```

推荐拆分：

| 层 | 职责 |
|---|---|
| 业务脚本 | 提取单据、构造业务报文、处理业务结果 |
| 连接器脚本 | 获取配置、获取 token、发送 HTTP、统一错误格式 |
| 工具函数 | `safeJsonParse`、日志、参数校验、签名辅助 |

## 通用连接器模板

```javascript
var AbstractAPIHandler = require('AbstractAPIHandler');

class MyAPIHandler extends AbstractAPIHandler {
    execute(request) {
        try {
            this.log('入参 request', request);
            if (!request || !request.apiPath) {
                throw new Error('apiPath不能为空');
            }
            var result = this.callThirdParty(request);
            return { success: true, data: result };
        } catch (e) {
            this.log('执行异常', e.message);
            return { success: false, message: e.message };
        }
    }
    log(message, data) {
        if (data !== undefined) {
            console.error('【MyAPIHandler】' + message + ':' + JSON.stringify(data));
        } else {
            console.error('【MyAPIHandler】' + message);
        }
    }
    requireValue(value, name) {
        if (value === null || value === undefined || value === '') {
            throw new Error('配置缺失：' + name);
        }
        return value;
    }
    getConfig(systemType) {
        var env = ObjectStore.env();
        var prefix = systemType || 'sap';
        return {
            baseUrl: this.requireValue(env[prefix + 'BaseUrl'], prefix + 'BaseUrl'),
            tokenUrl: env[prefix + 'TokenUrl'],
            clientId: env[prefix + 'ClientId'],
            clientSecret: env[prefix + 'ClientSecret'],
            apiKey: env[prefix + 'ApiKey']
        };
    }
    getToken(config) {
        if (config.apiKey) {
            return config.apiKey;
        }
        if (!config.tokenUrl) {
            return null;
        }
        var header = { 'Content-Type': 'application/x-www-form-urlencoded' };
        var body = {
            grant_type: 'client_credentials',
            client_id: config.clientId,
            client_secret: config.clientSecret
        };
        var resultString = ObjectStore.postman('POST', config.tokenUrl, header, body, 'form');
        var result = safeJsonParse(resultString, { longFields: [], lengthThreshold: 16 });
        return result.access_token || result.token || result.data;
    }
    buildHeader(token, customHeader) {
        var header = { 'Content-Type': 'application/json;charset=UTF-8' };
        if (token) {
            header.Authorization = 'Bearer ' + token;
        }
        if (customHeader) {
            for (var key in customHeader) {
                if (customHeader.hasOwnProperty(key)) {
                    header[key] = customHeader[key];
                }
            }
        }
        return header;
    }
    callThirdParty(request) {
        var config = this.getConfig(request.systemType);
        var token = this.getToken(config);
        var header = this.buildHeader(token, request.header);
        var url = config.baseUrl + request.apiPath;
        this.log('三方请求URL', url);
        var resultString = ObjectStore.postman(request.method || 'POST', url, header, request.body || {});
        var result = safeJsonParse(resultString, {
            longFields: request.longFields || [],
            lengthThreshold: 16
        });
        this.log('三方响应', result);
        return result;
    }
}

exports({ entryPoint: MyAPIHandler });
```

## NCC 连接器差异

### 已有现场封装优先

```javascript
function invokeNcc(nccRequest, uri) {
    console.error('【MyTrigger】开始调用NCC脚本:RBSM.APIhs.InvokeNCC');
    var func = extrequire('RBSM.APIhs.InvokeNCC');
    var invokeParam = { key: JSON.stringify(nccRequest), uri: uri };
    var result = func.execute(invokeParam);
    if (typeof result === 'string') {
        return safeJsonParse(result, { longFields: [], lengthThreshold: 16 });
    }
    return result;
}
```

### 无现场封装时的基础调用

```javascript
function getNccEnv(requestEnv) {
    if (requestEnv) {
        return requestEnv;
    }
    var env = ObjectStore.env();
    return {
        baseUrl: env.nccBaseUrl,
        clientId: env.nccClientId,
        clientSecret: env.nccClientSecret,
        pubKey: env.nccPubKey,
        grantType: env.nccGrantType || 'client_credentials',
        tokenUrl: env.nccBaseUrl + '/nccloud/opm/accesstoken'
    };
}

function callNcc(request) {
    var nccEnv = getNccEnv(request.nccEnv);
    var url = nccEnv.baseUrl + request.apiPath;
    return ObjectStore.nccLinker(
        request.method || 'POST',
        url,
        { 'Content-Type': 'application/json;charset=utf-8' },
        request.body || {},
        nccEnv
    );
}
```

## 业务脚本调用连接器

```javascript
function callSAP(apiPath, body) {
    var connector = extrequire('GTxxx.Integration.ThirdPartyConnector');
    var result = connector.execute({
        systemType: 'sap',
        apiPath: apiPath,
        method: 'POST',
        body: body,
        longFields: ['id', 'orgId', 'materialId']
    });
    if (!result || !result.success) {
        throw new Error('SAP调用失败：' + (result && result.message ? result.message : '未知错误'));
    }
    return result.data;
}
```

## 配置规范

- 认证信息必须来自 `ObjectStore.env()` 或平台配置中心，禁止硬编码到脚本。
- `baseUrl`、`clientSecret`、`apiKey` 等敏感值禁止写入日志。
- 不同系统使用不同前缀，例如 `sapBaseUrl`、`sapTokenUrl`、`sapClientId`、`sapClientSecret`。
- 三方返回涉及业务主键/外键时，必须按 [../10-common/04_Long精度防御.md](../10-common/04_Long精度防御.md) 使用 `safeJsonParse`。

## 自检清单

- 是否固定入口类 `MyAPIHandler` 或 `MyTrigger`。
- 是否遵循平台模板和项目风格使用 `var`/`let`，未使用 `const`、箭头函数，类内方法调用均带 `this.`。
- 是否从配置中心读取认证信息。
- 是否使用 `ObjectStore.postman` 或 `ObjectStore.nccLinker`。
- 是否对返回 JSON 使用 `safeJsonParse`。
- 是否记录请求 URL、响应摘要和异常，但不记录密钥。
